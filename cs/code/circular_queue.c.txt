global TheArray,HeadPointer,TailPointer,max,value
TheArray = [""] * 5
HeadPointer = 0
TailPointer = 0
max = 5

def Enqueue(value):
    global TheArray,HeadPointer,TailPointer,max
    if (HeadPointer == TailPointer) and (TheArray[HeadPointer] != ""):
        print('Array Full, Cannot Add')
    else:
        TheArray[TailPointer] = value
        print(TheArray)
        TailPointer += 1
        if TailPointer == max:
            TailPointer = 0

    global TheArray,HeadPointer,TailPointer,max
    if (HeadPointer==TailPointer) and (TheArray[HeadPointer] == ""):
        print("Empty, so we cant remove")

        removed = TheArray[HeadPointer]
        TheArray[HeadPointer] = ""
